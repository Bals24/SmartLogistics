-- Nettoyage si nécessaire
DROP TABLE IF EXISTS maintenance CASCADE;
DROP TABLE IF EXISTS incidents CASCADE;
DROP TABLE IF EXISTS livraisons CASCADE;
DROP TABLE IF EXISTS itineraire_etapes CASCADE;
DROP TABLE IF EXISTS itineraires CASCADE;
DROP TABLE IF EXISTS colis CASCADE;
DROP TABLE IF EXISTS commandes CASCADE;
DROP TABLE IF EXISTS vehicules CASCADE;
DROP TABLE IF EXISTS chauffeurs CASCADE;
DROP TABLE IF EXISTS entrepots CASCADE;
DROP TABLE IF EXISTS clients CASCADE;


-- ============================================================
-- 1. CLIENTS
-- ============================================================

CREATE TABLE clients (
    id BIGSERIAL PRIMARY KEY,
    nom VARCHAR(100) NOT NULL,
    prenom VARCHAR(100),
    telephone VARCHAR(30),
    email VARCHAR(150) UNIQUE,
    adresse TEXT,
    ville VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


-- ============================================================
-- 2. ENTREPÔTS
-- ============================================================

CREATE TABLE entrepots (
    id BIGSERIAL PRIMARY KEY,
    nom VARCHAR(150) NOT NULL,
    adresse TEXT,
    ville VARCHAR(100),
    capacite INTEGER NOT NULL DEFAULT 0,
    stock_actuel INTEGER NOT NULL DEFAULT 0,
    statut VARCHAR(30) NOT NULL DEFAULT 'ACTIF',

    CONSTRAINT chk_entrepot_capacite
        CHECK (capacite >= 0),

    CONSTRAINT chk_entrepot_stock
        CHECK (stock_actuel >= 0 AND stock_actuel <= capacite)
);


-- ============================================================
-- 3. VÉHICULES
-- ============================================================

CREATE TABLE vehicules (
    id BIGSERIAL PRIMARY KEY,
    immatriculation VARCHAR(30) NOT NULL UNIQUE,
    marque VARCHAR(80),
    modele VARCHAR(80),
    type VARCHAR(50),
    capacite_kg NUMERIC(10,2) NOT NULL DEFAULT 0,
    kilometrage NUMERIC(12,2) NOT NULL DEFAULT 0,
    statut VARCHAR(30) NOT NULL DEFAULT 'DISPONIBLE',

    CONSTRAINT chk_vehicule_capacite
        CHECK (capacite_kg >= 0),

    CONSTRAINT chk_vehicule_kilometrage
        CHECK (kilometrage >= 0)
);


-- ============================================================
-- 4. CHAUFFEURS
-- ============================================================

CREATE TABLE chauffeurs (
    id BIGSERIAL PRIMARY KEY,
    nom VARCHAR(100) NOT NULL,
    prenom VARCHAR(100),
    telephone VARCHAR(30),
    numero_permis VARCHAR(80) NOT NULL UNIQUE,
    type_permis VARCHAR(30),
    date_embauche DATE,
    statut VARCHAR(30) NOT NULL DEFAULT 'DISPONIBLE'
);


-- ============================================================
-- 5. COMMANDES
-- ============================================================

CREATE TABLE commandes (
    id BIGSERIAL PRIMARY KEY,
    reference VARCHAR(50) NOT NULL UNIQUE,
    client_id BIGINT NOT NULL,
    entrepot_id BIGINT,

    date_commande TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    date_livraison_prevue TIMESTAMP,

    adresse_livraison TEXT NOT NULL,
    ville_livraison VARCHAR(100),

    statut VARCHAR(30) NOT NULL DEFAULT 'EN_ATTENTE',
    montant NUMERIC(12,2) DEFAULT 0,

    CONSTRAINT fk_commande_client
        FOREIGN KEY (client_id)
        REFERENCES clients(id)
        ON DELETE RESTRICT,

    CONSTRAINT fk_commande_entrepot
        FOREIGN KEY (entrepot_id)
        REFERENCES entrepots(id)
        ON DELETE SET NULL,

    CONSTRAINT chk_commande_montant
        CHECK (montant >= 0)
);


-- ============================================================
-- 6. COLIS
-- ============================================================

CREATE TABLE colis (
    id BIGSERIAL PRIMARY KEY,
    reference VARCHAR(50) NOT NULL UNIQUE,
    commande_id BIGINT NOT NULL,

    description TEXT,
    poids_kg NUMERIC(10,2) NOT NULL DEFAULT 0,
    longueur_cm NUMERIC(10,2),
    largeur_cm NUMERIC(10,2),
    hauteur_cm NUMERIC(10,2),

    statut VARCHAR(30) NOT NULL DEFAULT 'EN_ATTENTE',

    CONSTRAINT fk_colis_commande
        FOREIGN KEY (commande_id)
        REFERENCES commandes(id)
        ON DELETE CASCADE,

    CONSTRAINT chk_colis_poids
        CHECK (poids_kg >= 0),

    CONSTRAINT chk_colis_dimensions
        CHECK (
            (longueur_cm IS NULL OR longueur_cm >= 0)
            AND
            (largeur_cm IS NULL OR largeur_cm >= 0)
            AND
            (hauteur_cm IS NULL OR hauteur_cm >= 0)
        )
);


-- ============================================================
-- 7. ITINÉRAIRES
-- ============================================================

CREATE TABLE itineraires (
    id BIGSERIAL PRIMARY KEY,
    nom VARCHAR(150) NOT NULL,
    origine VARCHAR(150) NOT NULL,
    destination VARCHAR(150) NOT NULL,

    distance_km NUMERIC(10,2) NOT NULL DEFAULT 0,
    duree_minutes INTEGER NOT NULL DEFAULT 0,

    statut VARCHAR(30) NOT NULL DEFAULT 'ACTIF',

    CONSTRAINT chk_itineraire_distance
        CHECK (distance_km >= 0),

    CONSTRAINT chk_itineraire_duree
        CHECK (duree_minutes >= 0)
);


-- ============================================================
-- 8. ÉTAPES D'ITINÉRAIRES
-- ============================================================

CREATE TABLE itineraire_etapes (
    id BIGSERIAL PRIMARY KEY,
    itineraire_id BIGINT NOT NULL,

    ordre INTEGER NOT NULL,
    nom_point VARCHAR(150) NOT NULL,

    latitude NUMERIC(10,7),
    longitude NUMERIC(10,7),

    CONSTRAINT fk_etape_itineraire
        FOREIGN KEY (itineraire_id)
        REFERENCES itineraires(id)
        ON DELETE CASCADE,

    CONSTRAINT uq_etape_ordre
        UNIQUE (itineraire_id, ordre),

    CONSTRAINT chk_etape_ordre
        CHECK (ordre > 0)
);


-- ============================================================
-- 9. LIVRAISONS
-- ============================================================

CREATE TABLE livraisons (
    id BIGSERIAL PRIMARY KEY,

    commande_id BIGINT NOT NULL,
    chauffeur_id BIGINT,
    vehicule_id BIGINT,
    itineraire_id BIGINT,

    date_depart TIMESTAMP,
    date_livraison TIMESTAMP,

    statut VARCHAR(30) NOT NULL DEFAULT 'PLANIFIEE',

    adresse_depart TEXT,
    adresse_arrivee TEXT,

    CONSTRAINT fk_livraison_commande
        FOREIGN KEY (commande_id)
        REFERENCES commandes(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_livraison_chauffeur
        FOREIGN KEY (chauffeur_id)
        REFERENCES chauffeurs(id)
        ON DELETE SET NULL,

    CONSTRAINT fk_livraison_vehicule
        FOREIGN KEY (vehicule_id)
        REFERENCES vehicules(id)
        ON DELETE SET NULL,

    CONSTRAINT fk_livraison_itineraire
        FOREIGN KEY (itineraire_id)
        REFERENCES itineraires(id)
        ON DELETE SET NULL
);


-- ============================================================
-- 10. INCIDENTS
-- ============================================================

CREATE TABLE incidents (
    id BIGSERIAL PRIMARY KEY,

    livraison_id BIGINT,
    vehicule_id BIGINT,
    chauffeur_id BIGINT,

    type VARCHAR(100) NOT NULL,
    description TEXT NOT NULL,

    gravite VARCHAR(30) NOT NULL DEFAULT 'MOYENNE',
    statut VARCHAR(30) NOT NULL DEFAULT 'OUVERT',

    date_incident TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    date_resolution TIMESTAMP,

    CONSTRAINT fk_incident_livraison
        FOREIGN KEY (livraison_id)
        REFERENCES livraisons(id)
        ON DELETE SET NULL,

    CONSTRAINT fk_incident_vehicule
        FOREIGN KEY (vehicule_id)
        REFERENCES vehicules(id)
        ON DELETE SET NULL,

    CONSTRAINT fk_incident_chauffeur
        FOREIGN KEY (chauffeur_id)
        REFERENCES chauffeurs(id)
        ON DELETE SET NULL
);


-- ============================================================
-- 11. MAINTENANCE
-- ============================================================

CREATE TABLE maintenance (
    id BIGSERIAL PRIMARY KEY,

    vehicule_id BIGINT NOT NULL,

    type VARCHAR(100) NOT NULL,
    description TEXT,

    date_planifiee DATE,
    date_realisation DATE,

    cout NUMERIC(12,2) DEFAULT 0,

    statut VARCHAR(30) NOT NULL DEFAULT 'PLANIFIEE',

    CONSTRAINT fk_maintenance_vehicule
        FOREIGN KEY (vehicule_id)
        REFERENCES vehicules(id)
        ON DELETE CASCADE,

    CONSTRAINT chk_maintenance_cout
        CHECK (cout >= 0)
);


-- ============================================================
-- INDEX
-- ============================================================

CREATE INDEX idx_commandes_client
    ON commandes(client_id);

CREATE INDEX idx_commandes_entrepot
    ON commandes(entrepot_id);

CREATE INDEX idx_colis_commande
    ON colis(commande_id);

CREATE INDEX idx_livraisons_commande
    ON livraisons(commande_id);

CREATE INDEX idx_livraisons_chauffeur
    ON livraisons(chauffeur_id);

CREATE INDEX idx_livraisons_vehicule
    ON livraisons(vehicule_id);

CREATE INDEX idx_incidents_livraison
    ON incidents(livraison_id);

CREATE INDEX idx_incidents_vehicule
    ON incidents(vehicule_id);

CREATE INDEX idx_maintenance_vehicule
    ON maintenance(vehicule_id);
